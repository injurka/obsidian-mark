**Задача:** Создать Data Asset, который будет содержать всю информацию о конкретном типе Шасси, необходимую для выбора и спавна (имя, иконка, описание, класс пешки, начальные статы/способности).

    #include "GameFramework/PlayerController.h"

    void UFrameForgeGameInstance::StorePlayerChassisSelection(APlayerController* PlayerController, TSubclassOf<AFrameForgeCharacter> ChassisClass)
    {
        if (!PlayerController)
        {
            UE_LOG(LogTemp, Warning, TEXT("UFrameForgeGameInstance::StorePlayerChassisSelection: PlayerController is null."));
            return;
        }
        if (!IsValid(ChassisClass))
        {
            UE_LOG(LogTemp, Warning, TEXT("UFrameForgeGameInstance::StorePlayerChassisSelection: Invalid ChassisClass provided for PlayerController %s."), *PlayerController->GetName());
            // Возможно, стоит удалить старую запись, если она была
            PendingPlayerChassisSelections.Remove(PlayerController);
            return;
        }

        // Добавляем или обновляем запись в карте
        PendingPlayerChassisSelections.Add(PlayerController, ChassisClass);
        UE_LOG(LogTemp, Log, TEXT("UFrameForgeGameInstance: Stored chassis selection %s for %s"), *ChassisClass->GetName(), *PlayerController->GetName());
    }

    TSubclassOf<AFrameForgeCharacter> UFrameForgeGameInstance::GetStoredChassisSelection(APlayerController* PlayerController)
    {
        if (!PlayerController)
        {
            UE_LOG(LogTemp, Warning, TEXT("UFrameForgeGameInstance::GetStoredChassisSelection: PlayerController is null."));
            return nullptr;
        }

        // Ищем запись по контроллеру
        const TSubclassOf<AFrameForgeCharacter>* FoundClass = PendingPlayerChassisSelections.Find(PlayerController);

        if (FoundClass && IsValid(*FoundClass)) // Проверяем и указатель, и сам класс
        {
            UE_LOG(LogTemp, Log, TEXT("UFrameForgeGameInstance: Found stored chassis selection %s for %s"), *(*FoundClass)->GetName(), *PlayerController->GetName());
            return *FoundClass;
        }
        else
        {
            UE_LOG(LogTemp, Warning, TEXT("UFrameForgeGameInstance::GetStoredChassisSelection: No valid selection found for %s."), *PlayerController->GetName());
            return nullptr;
        }
    }

    void UFrameForgeGameInstance::ClearStoredSelections()
    {
        PendingPlayerChassisSelections.Empty();
        UE_LOG(LogTemp, Log, TEXT("UFrameForgeGameInstance: Cleared all stored chassis selections."));
    }

    // Реализация логики старта игры (вызывается из PlayerController)
    void AGM_Lobby::RequestStartGame(APlayerController* RequestingPlayer)
    {
        // Выполняем ТОЛЬКО НА СЕРВЕРЕ
        if (!HasAuthority())
        {
            return;
        }

        UE_LOG(LogTemp, Log, TEXT("AGM_Lobby: Received start request from %s."), *GetNameSafe(RequestingPlayer));

        // --- Опциональные Проверки ---
        // TODO: Проверить, является ли RequestingPlayer хостом (если нужно)
        // TODO: Проверить систему готовности игроков (если есть)

        // --- Получение GameInstance и GameState ---
        UFrameForgeGameInstance* GI = GetGameInstance<UFrameForgeGameInstance>();
        AGameStateBase* GS = GetGameState<AGameStateBase>(); // Используем базовый GameState

        if (!GI)
        {
            UE_LOG(LogTemp, Error, TEXT("AGM_Lobby::RequestStartGame: Failed to get FrameForgeGameInstance!"));
            return;
        }
        if (!GS)
        {
            UE_LOG(LogTemp, Error, TEXT("AGM_Lobby::RequestStartGame: Failed to get GameStateBase!"));
            return;
        }

        // --- Подготовка к Переходу ---
        UE_LOG(LogTemp, Log, TEXT("AGM_Lobby: Preparing for Server Travel..."));

        // 1. Очистка старых данных в GameInstance
        GI->ClearStoredSelections();

        // 2. Сбор и сохранение выбора Шасси для КАЖДОГО игрока
        bool bAllPlayersReady = true; // Флаг для проверки, все ли выбрали
        for (APlayerState* PS_Generic : GS->PlayerArray)
        {
            AFrameForgePlayerState* PS = Cast<AFrameForgePlayerState>(PS_Generic);
            APlayerController* PC = PS ? PS->GetPlayerController() : nullptr; // Получаем контроллер из PlayerState

            if (PC && PS) // Убеждаемся, что и PS, и PC валидны
            {
                UChassisDataAsset* SelectedData = PS->GetSelectedChassisData();
                if (SelectedData && IsValid(SelectedData->ChassisPawnClass))
                {
                    // Сохраняем ВАЛИДНЫЙ выбор в GameInstance
                    GI->StorePlayerChassisSelection(PC, SelectedData->ChassisPawnClass);
                }
                else
                {
                    // Если выбор не сделан или невалиден
                    UE_LOG(LogTemp, Warning, TEXT("AGM_Lobby: Player %s has no valid chassis selected! Assigning default or aborting start? (Assigning default for now)"), *PS->GetPlayerName());
                    // Назначаем класс по умолчанию (из GameMode или другой логики)
                    TSubclassOf<AFrameForgeCharacter> DefaultChassisClass = TSubclassOf<AFrameForgeCharacter>(AFrameForgeCharacter::StaticClass()); // ЗАМЕНИТЕ НА ВАШ КЛАСС ПО УМОЛЧАНИЮ, ЕСЛИ НУЖНО
                    GI->StorePlayerChassisSelection(PC, DefaultChassisClass);

                    // Или можно установить флаг в false и прервать старт, если выбор обязателен
                    // bAllPlayersReady = false;
                    // break; // Выходим из цикла, если хотя бы один не готов/не выбрал
                }
            }
            else
            {
                 UE_LOG(LogTemp, Warning, TEXT("AGM_Lobby: Found invalid PlayerState or PlayerController in GameState->PlayerArray."));
                 bAllPlayersReady = false; // Считаем это ошибкой
                 // break;
            }
        }

        // 3. Проверка, все ли готовы (если нужно)
        if (!bAllPlayersReady)
        {
            UE_LOG(LogTemp, Warning, TEXT("AGM_Lobby: Not all players ready or have selected a valid chassis. Aborting game start."));
            // TODO: Отправить сообщение игрокам об ошибке
            return;
        }

        // --- Выполнение Server Travel ---
        UWorld* World = GetWorld();
        if (World)
        {
            // Формируем URL для перехода
            // bUseSeamlessTravel = true; // Устанавливаем в конструкторе
            FString URL = FirstGameMapPath; // Берем путь из переменной класса

            // Добавляем опцию ?listen для Listen Server
            if (GetNetMode() == NM_ListenServer)
            {
                URL += TEXT("?listen");
            }
            // Для Dedicated Server ?listen не нужен

            UE_LOG(LogTemp, Log, TEXT("AGM_Lobby: Executing ServerTravel to %s"), *URL);
            World->ServerTravel(URL, true); // true = Absolute (игнорируется, если путь начинается с /Game/), false = Relative
        }
        else
        {
             UE_LOG(LogTemp, Error, TEXT("AGM_Lobby::RequestStartGame: GetWorld() returned null!"));
        }
    }
