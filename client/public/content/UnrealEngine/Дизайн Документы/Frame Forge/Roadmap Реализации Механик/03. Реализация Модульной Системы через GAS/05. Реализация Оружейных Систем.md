**Реализация Оружейных Систем** с использованием GAS. Это ключевой этап для боевой системы.

**Задача:** Создать сменные основные атаки (Оружейные Системы), которые заменяют базовую атаку Шасси, имеют свою логику (снаряды/хитскан), урон, скорострельность и могут быть модифицированы.

**Зачем:** Реализовать динамичную смену основного оружия и его кастомизацию через модификаторы, что является важной частью GDD.

**GAS Компоненты и Механизмы UE:** `UGameplayAbility` (наследуемся от `FrameForgeGameplayAbility`), `GameplayEffect` (Damage, Cooldown), `GameplayTag`, `AbilityTask_WaitTargetData`, `SpawnActor`, `LineTraceByChannel`, `WeaponSystemItemData`, `WeaponModItemData` (новые), `ChassisSlotManagerComponent`.

    #pragma once

    #include "CoreMinimal.h"
    #include "Abilities/FrameForgeGameplayAbility.h"
    #include "FrameForgeWeaponAbility.generated.h"

    class UWeaponSystemItemData; // Прямое объявление

    UCLASS(Abstract)
    class FRAMEFORGE_API UFrameForgeWeaponAbility : public UFrameForgeGameplayAbility
    {
        GENERATED_BODY()

    public:
        UFrameForgeWeaponAbility();

        // --- Переопределения для Оружия ---

        // Проверяем дополнительные условия (патроны, состояние оружия)
        virtual bool CanActivateAbility(const FGameplayAbilitySpecHandle Handle, const FGameplayAbilityActorInfo* ActorInfo, const FGameplayTagContainer* SourceTags = nullptr, const FGameplayTagContainer* TargetTags = nullptr, OUT FGameplayTagContainer* OptionalRelevantTags = nullptr) const override;

        // Основная логика активации (вызывает Commit, затем PerformFireAction)
        virtual void ActivateAbility(const FGameplayAbilitySpecHandle Handle, const FGameplayAbilityActorInfo* ActorInfo, const FGameplayAbilityActivationInfo ActivationInfo, const FGameplayEventData* TriggerEventData) override;

    protected:
        // --- Функции для Наследников ---

        /** Основное действие выстрела (реализуется в дочерних BP/C++). Вызывается после успешного CommitAbility. */
        UFUNCTION(BlueprintImplementableEvent, Category = "Weapon Ability", meta = (DisplayName = "Perform Fire Action"))
        void K2_PerformFireAction();

        /** Возвращает Data Asset текущего оружия (если возможно). */
        UFUNCTION(BlueprintPure, Category = "Weapon Ability")
        UWeaponSystemItemData* GetWeaponItemData() const;

        /** Применяет эффект урона оружия к цели. */

        /** Применяет эффект урона оружия к целевому ASC. */
        UFUNCTION(BlueprintCallable, Category = "Weapon Ability")
        void ApplyWeaponDamageEffectToASC(TSubclassOf<UGameplayEffect> DamageEffect, UAbilitySystemComponent* TargetASC, const FGameplayTagContainer& OptionalSourceTags = FGameplayTagContainer());

        // --- Внутренние Функции ---

        /** Получает эффект урона из Data Asset оружия. */
        TSubclassOf<UGameplayEffect> GetWeaponDamageEffectClass() const;
    };

    void UFrameForgeWeaponAbility::ActivateAbility(const FGameplayAbilitySpecHandle Handle, const FGameplayAbilityActorInfo* ActorInfo, const FGameplayAbilityActivationInfo ActivationInfo, const FGameplayEventData* TriggerEventData)
    {
        // Сначала вызываем Super, чтобы применились ActivationOwnedTags
        Super::ActivateAbility(Handle, ActorInfo, ActivationInfo, TriggerEventData);

        // Пытаемся выполнить Commit (проверка блокирующих тегов, применение стоимости и кулдауна)
        if (CommitAbility(Handle, ActorInfo, ActivationInfo))
        {
            // Успешно! Вызываем основное действие выстрела (реализуется в дочерних классах)
            K2_PerformFireAction();
        }
        else
        {
            // Не удалось выполнить Commit (кулдаун, стоимость, блокирующий тег)
            // Завершаем способность с отменой
            constexpr bool bReplicateEndAbility = true;
            constexpr bool bWasCancelled = true;
            EndAbility(Handle, ActorInfo, ActivationInfo, bReplicateEndAbility, bWasCancelled);
        }
        // Не вызываем EndAbility здесь, если Commit успешен. Наследник должен вызвать EndAbility после завершения выстрела.
    }

     TSubclassOf<UGameplayEffect> UFrameForgeWeaponAbility::GetWeaponDamageEffectClass() const
     {

    void UFrameForgeWeaponAbility::ApplyWeaponDamageEffect(TSubclassOf<UGameplayEffect> DamageEffect, const FHitResult& HitResult, const FGameplayTagContainer& OptionalSourceTags)
    {
        AActor* TargetActor = HitResult.GetActor();
        if (!TargetActor) return;

        UAbilitySystemComponent* TargetASC = UAbilitySystemBlueprintLibrary::GetAbilitySystemComponent(TargetActor);
        ApplyWeaponDamageEffectToASC(DamageEffect, TargetASC, OptionalSourceTags);
    }

     void UFrameForgeWeaponAbility::ApplyWeaponDamageEffectToASC(TSubclassOf<UGameplayEffect> DamageEffect, UAbilitySystemComponent* TargetASC, const FGameplayTagContainer& OptionalSourceTags)
    {
        if (!TargetASC || !IsValid(DamageEffect)) return;

        UAbilitySystemComponent* SourceASC = GetAbilitySystemComponentFromActorInfo();
        if (!SourceASC) return;

        FGameplayEffectContextHandle ContextHandle = SourceASC->MakeEffectContext();
        ContextHandle.AddSourceObject(GetAvatarActorFromActorInfo()); // Источник - наш персонаж
        // ContextHandle.AddHitResult(HitResult); // Можно добавить HitResult в контекст

        FGameplayEffectSpecHandle SpecHandle = SourceASC->MakeOutgoingSpec(DamageEffect, GetAbilityLevel(), ContextHandle);
        if (SpecHandle.IsValid())
        {
            // Добавляем опциональные теги (например, для крита)
            SpecHandle.Data->AppendDynamicAssetTags(OptionalSourceTags);

            FActiveGameplayEffectHandle AppliedHandle = SourceASC->ApplyGameplayEffectSpecToTarget(*SpecHandle.Data.Get(), TargetASC);
            // UE_LOG(LogTemp, Verbose, TEXT("Applied damage effect %s to %s"), *DamageEffect->GetName(), *TargetASC->GetOwnerActor()->GetName());
        }
    }
