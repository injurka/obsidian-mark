## ✅ Готовность маршрутных дней

```dataviewjs
const routeFolder = dv.current().file.folder;
const days = dv.pages(`"${routeFolder}"`)
  .where((page) => typeof page.day === "number")
  .sort((page) => page.day, "asc");

const ready = days.where((page) => page.is_ready === true);
const invalid = days.where((page) => typeof page.is_ready !== "boolean");
const requiredFields = [
  "date",
  "weekday",
  "title",
  "location",
  "phase",
  "accommodation",
  "highlight",
  "is_ready",
  "mode",
  "base_city",
  "transport_types",
  "booking_status",
];
const metadataIssues = [];
for (const page of days) {
  for (const field of requiredFields) {
    if (page[field] === undefined || page[field] === null) {
      metadataIssues.push([page.day, field, page.file.link]);
    }
  }
}
const percent = days.length ? Math.round((ready.length / days.length) * 100) : 0;

dv.paragraph(`**${ready.length} из ${days.length} дней готовы (${percent}%)**`);
dv.paragraph(`<progress value="${ready.length}" max="${Math.max(days.length, 1)}"></progress>`);

const modeLabels = {
  transit: "🛫 Транзит",
  vacation: "🌴 Отпуск",
  workation: "💻 Воркейшн",
  island_weekend: "🐢 Островной уикенд",
};
dv.header(3, "Сводка по режимам");
dv.table(
  ["Режим", "Дней", "Готово"],
  Object.entries(modeLabels).map(([mode, label]) => [
    label,
    days.where((page) => page.mode === mode).length,
    days.where((page) => page.mode === mode && page.is_ready === true).length,
  ])
);

dv.header(3, "Маршрутные дни");
dv.table(
  ["День", "Дата", "Режим", "План", "База", "Статус"],
  days.map((page) => [
    page.day,
    dv.date(page.date).toFormat("dd.MM"),
    modeLabels[page.mode] ?? page.mode ?? "—",
    dv.fileLink(page.file.path, false, page.title ?? `День ${page.day}`),
    page.base_city ?? "—",
    page.is_ready === true ? "✅ Готов" : page.is_ready === false ? "⬜ Подготовить" : "❗ Проверить",
  ])
);

if (invalid.length || metadataIssues.length) {
  dv.header(3, "Ошибки метаданных");
  if (invalid.length) {
    dv.table(
      ["Проблема", "Файл"],
      invalid.map((page) => ["is_ready не boolean", page.file.link])
    );
  }
  if (metadataIssues.length) {
    dv.table(["День", "Отсутствующее поле", "Файл"], metadataIssues);
  }
}
```
