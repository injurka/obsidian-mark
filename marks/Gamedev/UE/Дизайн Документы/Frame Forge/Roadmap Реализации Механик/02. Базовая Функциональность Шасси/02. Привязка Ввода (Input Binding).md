**Привязка Ввода (Input Binding)** с использованием Enhanced Input и GAS.

**Задача:** Настроить систему Enhanced Input, создать необходимые Input Actions (IA) и Input Mapping Context (IMC). Затем, в коде персонажа, привязать срабатывание этих IA к вызову функций `AbilityLocalInputPressed`/`AbilityLocalInputReleased` у `AbilitySystemComponent`, передавая соответствующий идентификатор из нового перечисления `EAbilityInputID`.

**Зачем:** Это мост между физическим нажатием кнопки игроком и логической активацией слота способности в Gameplay Ability System. Без этой привязки GAS не будет знать, какую способность пытался активировать игрок.

**GAS Компоненты и Механизмы UE:** `UEnhancedInputComponent`, `UInputAction`, `UInputMappingContext`, `AbilitySystemComponent::AbilityLocalInputPressed`, `AbilitySystemComponent::AbilityLocalInputReleased`, `EAbilityInputID` (новое перечисление).

---

**Шаг 7.1: Создание Перечисления `EAbilityInputID` (C++)**

Это перечисление определит логические "слоты" для способностей, на которые мы будем ссылаться при привязке ввода и выдаче способностей.

1.  **Найдите или создайте** подходящий заголовочный файл для глобальных перечислений (например, `FrameForge/Source/FrameForge/Public/Core/FrameForgeTypes.h` или создайте новый файл `AbilityInputID.h` в `_Core/Data/Enums/`).
2.  **Добавьте следующий код** в этот файл:

    ```cpp
    // FrameForge/Source/FrameForge/Public/Core/AbilityInputID.h (Пример пути)

    #pragma once

    #include "AbilityInputID.generated.h" // Для UENUM

    // Определяет логические идентификаторы для привязки ввода к способностям GAS.
    // Значения (кроме None) должны быть > 0.
    UENUM(BlueprintType)
    enum class EAbilityInputID : uint8
    {
        // 0 = None
        None            UMETA(DisplayName = "None"),
        // 1 = Confirm
        Confirm         UMETA(DisplayName = "Confirm"), // Подтверждение (например, для способностей с таргетингом)
        // 2 = Cancel
        Cancel          UMETA(DisplayName = "Cancel"),  // Отмена (например, для способностей с таргетингом)
        // 3 = Основная Атака (ЛКМ / Правый Триггер)
        PrimaryAttack   UMETA(DisplayName = "Primary Attack"),
        // 4 = Альтернативная Атака / Прицеливание (ПКМ / Левый Триггер) - если нужно
        SecondaryAttack UMETA(DisplayName = "Secondary Attack / Aim"),
        // 5 = Первая Активная Способность (Q / Левый Бампер)
        Ability1        UMETA(DisplayName = "Ability 1"),
        // 6 = Вторая Активная Способность (E / Правый Бампер)
        Ability2        UMETA(DisplayName = "Ability 2"),
        // 7 = Утилитарная Способность (Shift / Лицевая кнопка X/Квадрат)
        Utility         UMETA(DisplayName = "Utility"),
        // 8 = Способность Передвижения (Пробел в воздухе / Лицевая кнопка A/Крест)
        MovementAbility UMETA(DisplayName = "Movement Ability"),
        // 9 = Прыжок (Пробел на земле) - Можно использовать стандартный или через GAS
        Jump            UMETA(DisplayName = "Jump"),
        // 10 = Взаимодействие (F / Лицевая кнопка Y/Треугольник)
        Interact        UMETA(DisplayName = "Interact"),
        // 11 = Перезарядка (R / Лицевая кнопка B/Круг)
        Reload          UMETA(DisplayName = "Reload")
        // Добавьте другие по необходимости (Спринт, Приседание, если они будут через GAS)
    };
    ```
3.  **Сохраните файл.** Перекомпилируйте проект, чтобы редактор распознал новое перечисление.

    ```cpp
    void AFrameForgeCharacter::BindASCInput()
    {
        // Проверяем, что привязка еще не была выполнена
        if (bASCInputBound)
        {
            return;
        }

        // Получаем Enhanced Input Component
        if (UEnhancedInputComponent* EnhancedInputComponent = Cast<UEnhancedInputComponent>(InputComponent))
        {
            // Получаем Ability System Component (он должен быть уже инициализирован к этому моменту)
            UAbilitySystemComponent* ASC = GetAbilitySystemComponent(); // Используем функцию интерфейса
            if (!ASC)
            {
                UE_LOG(LogTemp, Error, TEXT("AFrameForgeCharacter::BindASCInput: Cannot find AbilitySystemComponent for %s"), *GetName());
                return; // Не можем привязать без ASC
            }

            // --- Привязка Input Actions к вызовам AbilityLocalInputPressed/Released ---

            // Макрос для упрощения привязки
            #define BIND_ABILITY_ACTION(InputACTION, InputID) \
            if (InputACTION) { \
                EnhancedInputComponent->BindAction(InputACTION, ETriggerEvent::Triggered, this, &AFrameForgeCharacter::Input_AbilityInputTagPressed, EAbilityInputID::InputID); \
                EnhancedInputComponent->BindAction(InputACTION, ETriggerEvent::Completed, this, &AFrameForgeCharacter::Input_AbilityInputTagReleased, EAbilityInputID::InputID); \
                 UE_LOG(LogTemp, Verbose, TEXT("Bound Input Action %s to Input ID %d"), *InputACTION->GetName(), static_cast<int32>(EAbilityInputID::InputID)); \
            } else { \
                 UE_LOG(LogTemp, Warning, TEXT("Input Action for Input ID %d is not set in %s!"), static_cast<int32>(EAbilityInputID::InputID), *GetName()); \
            }

            // Используем макрос для каждой привязки
            BIND_ABILITY_ACTION(Input_PrimaryAttack, PrimaryAttack);
            BIND_ABILITY_ACTION(Input_SecondaryAttack, SecondaryAttack);
            BIND_ABILITY_ACTION(Input_Ability1, Ability1);
            BIND_ABILITY_ACTION(Input_Ability2, Ability2);
            BIND_ABILITY_ACTION(Input_Utility, Utility);
            BIND_ABILITY_ACTION(Input_MovementAbility, MovementAbility);
            BIND_ABILITY_ACTION(Input_Jump, Jump); // Привязываем, даже если используем стандартный прыжок (GAS может его переопределить)
            BIND_ABILITY_ACTION(Input_Interact, Interact);
            BIND_ABILITY_ACTION(Input_Reload, Reload);
            BIND_ABILITY_ACTION(Input_ConfirmTarget, Confirm);
            BIND_ABILITY_ACTION(Input_CancelTarget, Cancel);

            #undef BIND_ABILITY_ACTION // Удаляем макрос после использования

            bASCInputBound = true; // Устанавливаем флаг
            UE_LOG(LogTemp, Log, TEXT("AFrameForgeCharacter::BindASCInput: Input bound to ASC for %s"), *GetName());
        }
         else
         {
              UE_LOG(LogTemp, Error, TEXT("AFrameForgeCharacter::BindASCInput: EnhancedInputComponent is null for %s"), *GetName());
         }
    }
    ```
