Сейчас мы создадим базовые ресурсы, которые можно найти в мире, и свяжем их с системой строительства, чтобы постройка стала не бесплатной.

### **Шаг 3.3: Создание базовых строительных элементов (ветки, глина)**

**Задача:**
1.  **Создать Систему Предметов (Items):** Нам нужна архитектура для описания любого предмета в игре — его название, иконка, можно ли его стакать и т.д.
2.  **Создать Ресурсы:** Определить конкретные предметы: "Ветки" (Wood Branches) и "Глина" (Clay).
3.  **Создать Инвентарь:** Реализовать компонент инвентаря, который сможет хранить эти предметы.
4.  **Интегрировать в Строительство:** Изменить систему строительства так, чтобы она требовала ресурсы из инвентаря игрока для постройки объектов.

**Философия:**
1.  **Data-Driven (Управляемый Данными):** Мы не будем создавать C++ класс для каждого предмета. Вместо этого, мы создадим один базовый C++ класс `UItemDataAsset`, а все конкретные предметы ("Ветки", "Глина", "Камень") будут экземплярами этого `Data Asset`. Это позволяет гейм-дизайнерам легко добавлять сотни новых предметов, не написав ни строчки кода.
2.  **Компонент Инвентаря:** Логику хранения, добавления и удаления предметов мы вынесем в отдельный `UInventoryComponent`, который прикрепим к нашему персонажу. Это делает систему переиспользуемой (например, для сундуков).
3.  **Структура для Слотов:** Инвентарь будет хранить массив структур `FInventorySlot`, где каждая структура содержит ссылку на `Data Asset` предмета и его количество.
4.  **Проверка Ресурсов:** `BuildingComponent` перед отправкой запроса на сервер будет проверять в инвентаре игрока, достаточно ли у него ресурсов. Если да, то сервер при постройке спишет эти ресурсы.

    *   **В `InventoryComponent.h`:**

        #pragma once
        
        #include "CoreMinimal.h"
        #include "Components/ActorComponent.h"
        #include "Core/FF_Types.h"
        #include "InventoryComponent.generated.h"

        class FAUNALOCUS_API UInventoryComponent : public UActorComponent
        {

        public:	
        	UInventoryComponent();

        	UFUNCTION(BlueprintCallable, Category = "Inventory")

        protected:

        	TArray<FInventorySlot> InventoryContents;
        
        	UPROPERTY(EditDefaultsOnly, Category = "Inventory")
        	int32 InventorySize = 20;
        
        	UFUNCTION()
        	void OnRep_InventoryContents();
        
        	virtual void GetLifetimeReplicatedProps(TArray<FLifetimeProperty>& OutLifetimeProps) const override;
        };

    *   **В `InventoryComponent.cpp`:**
        ```cpp
        // FaunaLocus/Source/FaunaLocus/GameplaySystems/InventorySystem/InventoryComponent.cpp
        #include "GameplaySystems/InventorySystem/InventoryComponent.h"
        #include "GameplaySystems/InventorySystem/ItemDataAsset.h"
        #include "Net/UnrealNetwork.h"
        
        UInventoryComponent::UInventoryComponent()
        {
            PrimaryComponentTick.bCanEverTick = false;
            SetIsReplicatedByDefault(true);
        }
        
        bool UInventoryComponent::AddItem(UItemDataAsset* Item, int32 Quantity)
        {
            if (!GetOwner()->HasAuthority() || !Item || Quantity <= 0) return false;
        
            // TODO: Реализовать логику поиска свободного слота или стака
            // Пока что просто добавляем в новый слот для простоты
            if (InventoryContents.Num() < InventorySize)
            {
                FInventorySlot NewSlot;
                NewSlot.ItemData = Item;
                NewSlot.Quantity = Quantity;
                InventoryContents.Add(NewSlot);
                OnInventoryUpdated.Broadcast();
                return true;
            }
            return false;
        }
        
        bool UInventoryComponent::RemoveItem(UItemDataAsset* Item, int32 Quantity)
        {
            if (!GetOwner()->HasAuthority() || !Item || Quantity <= 0) return false;
            
            // Ищем предмет и удаляем нужное количество
            for (FInventorySlot& Slot : InventoryContents)
            {
                if (Slot.ItemData == Item)
                {
                    if (Slot.Quantity >= Quantity)
                    {
                        Slot.Quantity -= Quantity;
                        if (Slot.Quantity <= 0)
                        {
                            // Удаляем слот, если он опустел
                            InventoryContents.RemoveAll([&](const FInventorySlot& InSlot){ return InSlot.Quantity <= 0; });
                        }
                        OnInventoryUpdated.Broadcast();
                        return true;
                    }
                }
            }
            return false;
        }
        
        bool UInventoryComponent::HasItem(UItemDataAsset* Item, int32 Quantity) const
        {
            int32 TotalQuantity = 0;
            for (const FInventorySlot& Slot : InventoryContents)
            {
                if (Slot.ItemData == Item)
                {
                    TotalQuantity += Slot.Quantity;
                }
            }
            return TotalQuantity >= Quantity;
        }
        
        void UInventoryComponent::OnRep_InventoryContents()
        {
            OnInventoryUpdated.Broadcast();
        }
        
        void UInventoryComponent::GetLifetimeReplicatedProps(TArray<FLifetimeProperty>& OutLifetimeProps) const
        {
            Super::GetLifetimeReplicatedProps(OutLifetimeProps);
            DOREPLIFETIME(UInventoryComponent, InventoryContents);
        }
        ```
        *Внимание: Логика `AddItem` здесь упрощена для примера. Полноценная система должна искать существующие стаки того же предмета.*

        void UBuildingComponent::PlaceBuildingPiece()
        {
            if (bIsInBuildMode && GhostBuildingPiece && GhostBuildingPiece->CanBePlaced())

                UInventoryComponent* InventoryComp = GetOwner()->FindComponentByClass<UInventoryComponent>();

                    }

            ```cpp
            void UBuildingComponent::Server_PlaceBuildingPiece_Implementation(TSubclassOf<ABuildingPieceBase> ClassToSpawn, FTransform SpawnTransform, const TArray<FInventorySlot>& Cost)
            {
                UInventoryComponent* InventoryComp = GetOwner()->FindComponentByClass<UInventoryComponent>();
                if (!ClassToSpawn || !InventoryComp) return;
            
                // Повторная проверка на сервере (анти-чит)
                bool bHasEnoughResources = true;
                for (const FInventorySlot& Slot : Cost)
                {
                    if (!InventoryComp->HasItem(Slot.ItemData, Slot.Quantity))
                    {
                        bHasEnoughResources = false;
                        break;
                    }
                }
            
                if (bHasEnoughResources)
                {
                    // Списываем ресурсы
                    for (const FInventorySlot& Slot : Cost)
                    {
                        InventoryComp->RemoveItem(Slot.ItemData, Slot.Quantity);
                    }
            
                    // Строим здание
                    FActorSpawnParameters SpawnParams;
                    SpawnParams.SpawnCollisionHandlingOverride = ESpawnActorCollisionHandlingMethod::AdjustIfPossibleButAlwaysSpawn;
                    GetWorld()->SpawnActor<ABuildingPieceBase>(ClassToSpawn, SpawnTransform, SpawnParams);
                }
            }
            ```
