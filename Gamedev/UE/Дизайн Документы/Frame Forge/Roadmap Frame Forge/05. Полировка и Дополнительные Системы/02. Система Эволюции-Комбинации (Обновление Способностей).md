**Система Эволюции/Комбинации (Обновление Способностей/Эффектов)**. Мы создадим механизм, который атомарно заменяет способности и эффекты от исходных предметов на те, что предоставляет новый, созданный предмет.

**Задача:** При Эволюции/Комбинации предмета система должна корректно удалить старые способности/эффекты, связанные с исходными предметами, и выдать новые способности/эффекты от созданного предмета. Использовать `GameplayTag` (`ItemSourceTag`) для идентификации и надежного удаления пассивных эффектов.

**Зачем:** Реализовать ядро прогрессивной системы предметов, где найденные предметы не просто накапливаются, а трансформируются, меняя возможности Шасси.

**GAS Компоненты и Механизмы UE:** `UAbilitySystemComponent`, `ClearAbility`, `RemoveActiveGameplayEffectsWithGrantedTags`, `GiveAbility`, `ApplyGameplayEffectToSelf`, `GameplayTag`, `UDataAsset` (для рецептов), `ChassisSlotManagerComponent` (расширяем его функционал).

    ```cpp
    bool UChassisSlotManagerComponent::PerformItemUpgrade(UFrameForgeRecipeData* Recipe, const TArray<ESlotType>& IngredientSlotTypes, UFrameForgeItemData*& OutResultingItemData, ESlotType& OutTargetSlot)
    {
        OutResultingItemData = nullptr;
        OutTargetSlot = ESlotType::None;

        // --- 1. Проверки (Выполняем ТОЛЬКО НА СЕРВЕРЕ) ---
        AActor* Owner = GetOwner();
        if (!Owner || Owner->GetLocalRole() != ROLE_Authority)
        {
            UE_LOG(LogTemp, Warning, TEXT("PerformItemUpgrade: Called on client for %s. Ignoring."), *GetNameSafe(Owner));
            return false;
        }
        if (!Recipe || !Recipe->ResultingItem)
        {
            UE_LOG(LogTemp, Warning, TEXT("PerformItemUpgrade: Invalid Recipe or ResultingItem is null."));
            return false;
        }
        if (IngredientSlotTypes.Num() == 0)
        {
            UE_LOG(LogTemp, Warning, TEXT("PerformItemUpgrade: IngredientSlotTypes array is empty."));
            return false;
        }

        // --- 2. Проверка Ингредиентов ---
        TArray<UFrameForgeItemData*> FoundIngredientsData;
        TArray<ESlotType> ValidatedIngredientSlots; // Слоты, которые будем очищать

        if (Recipe->RecipeType == ERecipeType::Evolution)
        {
            // Эволюция: ищем нужное количество ОДИНАКОВЫХ предметов
            if (Recipe->Ingredients.Num() != 1 || !Recipe->Ingredients[0])
            {
                UE_LOG(LogTemp, Warning, TEXT("PerformItemUpgrade (Evolution): Recipe must have exactly one valid ingredient defined. Recipe: %s"), *Recipe->GetName());
                return false;
            }
            UFrameForgeItemData* RequiredIngredient = Recipe->Ingredients[0];
            int32 FoundCount = 0;
            for (ESlotType SlotType : IngredientSlotTypes)
            {
                if (const FEquippedSlotInfo* SlotInfo = EquippedSlots.Find(SlotType))
                {
                    if (SlotInfo->ItemData == RequiredIngredient)
                    {
                        FoundCount++;
                        FoundIngredientsData.Add(SlotInfo->ItemData); // Сохраняем для информации (хотя они одинаковые)
                        ValidatedIngredientSlots.Add(SlotType);       // Этот слот будет очищен
                        if (FoundCount >= Recipe->EvolutionIngredientCount)
                        {
                            break; // Нашли достаточно
                        }
                    }
                }
            }
            if (FoundCount < Recipe->EvolutionIngredientCount)
            {
                UE_LOG(LogTemp, Log, TEXT("PerformItemUpgrade (Evolution): Not enough ingredients found. Required %d of %s, Found %d."), Recipe->EvolutionIngredientCount, *RequiredIngredient->GetName(), FoundCount);
                return false;
            }
            // Оставляем в ValidatedIngredientSlots только нужное количество слотов
            ValidatedIngredientSlots.SetNum(Recipe->EvolutionIngredientCount);
        }
        else // ERecipeType::Combination
        {
            // Комбинация: проверяем точное соответствие и порядок (или без порядка, если нужно сложнее)
            if (IngredientSlotTypes.Num() != Recipe->Ingredients.Num())
            {
                UE_LOG(LogTemp, Warning, TEXT("PerformItemUpgrade (Combination): IngredientSlotTypes count (%d) does not match Recipe ingredient count (%d). Recipe: %s"), IngredientSlotTypes.Num(), Recipe->Ingredients.Num(), *Recipe->GetName());
                return false;
            }
            bool bAllMatch = true;
            for (int32 i = 0; i < Recipe->Ingredients.Num(); ++i)
            {
                UFrameForgeItemData* RequiredIngredient = Recipe->Ingredients[i];
                ESlotType CurrentSlot = IngredientSlotTypes[i];
                const FEquippedSlotInfo* SlotInfo = EquippedSlots.Find(CurrentSlot);

                if (!RequiredIngredient || !SlotInfo || SlotInfo->ItemData != RequiredIngredient)
                {
                    UE_LOG(LogTemp, Warning, TEXT("PerformItemUpgrade (Combination): Ingredient mismatch at index %d. Required: %s, Found in Slot %d: %s. Recipe: %s"),
                        i, *GetNameSafe(RequiredIngredient), (int32)CurrentSlot, *GetNameSafe(SlotInfo ? SlotInfo->ItemData : nullptr), *Recipe->GetName());
                    bAllMatch = false;
                    break;
                }
                FoundIngredientsData.Add(SlotInfo->ItemData);
                ValidatedIngredientSlots.Add(CurrentSlot); // Все эти слоты будут очищены
            }
            if (!bAllMatch)
            {
                return false;
            }
        }

        // TODO: Проверка наличия Катализатора/Чертежа (если Recipe->RequiredCatalyst != nullptr)
        // Это потребует доступа к инвентарю или другому месту хранения катализаторов.
        // bool bHasCatalyst = CheckInventoryForItem(Recipe->RequiredCatalyst);
        // if (!bHasCatalyst) return false;

        // --- 3. Определяем Целевой Слот и Проверяем Совместимость ---
        // Используем первый валидный слот из списка ингредиентов как целевой
        OutTargetSlot = ValidatedIngredientSlots[0];
        if (Recipe->ResultingItem->CompatibleSlotType != OutTargetSlot)
        {
             // Пытаемся найти другой подходящий слот среди ингредиентов
             bool bFoundCompatibleSlot = false;
             for(ESlotType Slot : ValidatedIngredientSlots)
             {
                 if(Recipe->ResultingItem->CompatibleSlotType == Slot)
                 {
                     OutTargetSlot = Slot;
                     bFoundCompatibleSlot = true;
                     break;
                 }
             }
             if(!bFoundCompatibleSlot)
             {
                UE_LOG(LogTemp, Warning, TEXT("PerformItemUpgrade: Resulting item %s (Slot %d) is not compatible with any of the ingredient slots. Cannot equip."),
                    *Recipe->ResultingItem->GetName(), Recipe->ResultingItem->CompatibleSlotType);
                // Возможно, стоит попытаться найти ЛЮБОЙ свободный совместимый слот на Шасси?
                // if (!FindAvailableSlot(Recipe->ResultingItem->CompatibleSlotType, OutTargetSlot)) return false;
                return false; // Пока просто отменяем
             }
        }

        // --- 4. Выполнение Апгрейда ---

        // 4.1 Снимаем ВСЕ предметы-ингредиенты из их слотов
        TArray<UFrameForgeItemData*> UnequippedItems; // Сохраняем для возможного отката или информации
        for (ESlotType SlotToClear : ValidatedIngredientSlots)
        {
            UFrameForgeItemData* RemovedData = UnequipItem(SlotToClear); // Unequip удаляет GA/GE
            if (!RemovedData)
            {
                // Это не должно произойти, если проверка прошла, но на всякий случай
                 UE_LOG(LogTemp, Error, TEXT("PerformItemUpgrade: Failed to unequip item from validated ingredient slot %d! Aborting."), (int32)SlotToClear);
                 // TODO: Попытаться откатить предыдущие Unequip? Сложно. Проще прервать.
                 return false;
            }
            UnequippedItems.Add(RemovedData);
        }

        // TODO: Потребление Катализатора/Чертежа из инвентаря
        // ConsumeItemFromInventory(Recipe->RequiredCatalyst);

        // 4.2 Экипируем результирующий предмет в целевой слот
        bool bEquipSuccess = EquipItem(Recipe->ResultingItem, OutTargetSlot);

        if (!bEquipSuccess)
        {
             UE_LOG(LogTemp, Error, TEXT("PerformItemUpgrade: Failed to equip resulting item %s into slot %d! Upgrade failed."), *Recipe->ResultingItem->GetName(), (int32)OutTargetSlot);
             // TODO: Попытка отката - вернуть снятые ингредиенты? Очень сложно и рискованно.
             // Проще считать операцию неудавшейся.
             return false;
        }

        // --- 5. Успех ---
        OutResultingItemData = Recipe->ResultingItem;
        UE_LOG(LogTemp, Log, TEXT("PerformItemUpgrade: Successfully upgraded items in slots [%s] using recipe %s. Result %s equipped in slot %d."),
               *FString::JoinBy(ValidatedIngredientSlots, TEXT(", "), [](ESlotType E){ return FString::FromInt((int32)E); }), // Форматируем массив слотов
               *Recipe->GetName(),
               *OutResultingItemData->GetName(),
               (int32)OutTargetSlot);

        // (Опционально) Оповещаем другие системы
        // OnItemUpgradedDelegate.Broadcast(OutTargetSlot, FoundIngredientsData[0], OutResultingItemData); // Пример

        return true;
    }
    ```
