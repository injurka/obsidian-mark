**Система Выдачи Способностей/Эффектов Предметами**. Это ядро вашей модульной системы. Мы создадим компонент для управления слотами Шасси и механизм для предоставления/отзыва GAS способностей и эффектов при экипировке/снятии предметов.

**Задача:** Разработать механизм, при котором экипировка Предмета (Модуля, Оружия, Компонента) в слот Шасси вызывает выдачу (Grant) соответствующей `GameplayAbility` или `GameplayEffect` персонажу. При снятии предмета — способность/эффект удаляются.

**Зачем:** Реализация ключевой механики *Frame Forge* - динамическое получение возможностей от экипированных предметов.

**GAS Компоненты и Механизмы UE:** `UAbilitySystemComponent`, `UGameplayAbility`, `UGameplayEffect`, `UDataAsset`, `GiveAbility`, `ClearAbility`, `ApplyGameplayEffectToSelf`, `RemoveActiveGameplayEffectsWithGrantedTags`, `FGameplayAbilitySpec`, `FGameplayAbilitySpecHandle`, `FActiveGameplayEffectHandle`, `FGameplayTag`, `FGameplayTagContainer`.

    bool UChassisSlotManagerComponent::EquipItem(UFrameForgeItemData* ItemToEquip, ESlotType TargetSlot)
    {
        // Выполняем ТОЛЬКО НА СЕРВЕРЕ
        AActor* Owner = GetOwner();
        if (!Owner || Owner->GetLocalRole() != ROLE_Authority)
        {
            UE_LOG(LogTemp, Warning, TEXT("ChassisSlotManagerComponent::EquipItem called on client for %s. Ignoring."), *GetNameSafe(Owner));
            return false;
        }

        if (!ItemToEquip)
        {
            UE_LOG(LogTemp, Warning, TEXT("ChassisSlotManagerComponent::EquipItem: ItemToEquip is null."));
            return false;
        }

        // Проверяем совместимость слота
        if (ItemToEquip->CompatibleSlotType != TargetSlot)
        {
             UE_LOG(LogTemp, Warning, TEXT("ChassisSlotManagerComponent::EquipItem: Item %s is not compatible with slot type %d."), *ItemToEquip->GetName(), TargetSlot);
             return false;
        }

        // Проверяем, доступен ли слот на Шасси
        if (!HasSlotAvailable(TargetSlot))
        {
             UE_LOG(LogTemp, Warning, TEXT("ChassisSlotManagerComponent::EquipItem: Slot type %d is not available on this chassis."), TargetSlot);
             return false;
        }

        UAbilitySystemComponent* ASC = GetOwnerASC();
        if (!ASC)
        {
            UE_LOG(LogTemp, Error, TEXT("ChassisSlotManagerComponent::EquipItem: Owner ASC is null. Cannot equip item."));
            return false;
        }

        // --- Сначала снимаем старый предмет, если он есть ---
        UnequipItem(TargetSlot); // Unequip позаботится об удалении старых GA/GE

        // --- Создаем запись о новом предмете ---
        FEquippedSlotInfo NewSlotInfo;
        NewSlotInfo.ItemData = ItemToEquip;
        NewSlotInfo.ItemSourceTag = ItemToEquip->ItemSourceTag; // Запоминаем тег

        bool bSuccess = false;
        FGameplayTag GrantedWeaponTag; // Для хранения тега оружия

        // --- Выдаем Способность или Эффект в зависимости от типа предмета ---
        if (UModuleAbilityItemData* AbilityData = Cast<UModuleAbilityItemData>(ItemToEquip))
        {
            bSuccess = GrantAbilityToOwner(AbilityData, TargetSlot, NewSlotInfo.GrantedAbilityHandle);
        }
        else if (UComponentPassiveItemData* PassiveData = Cast<UComponentPassiveItemData>(ItemToEquip))
        {
            bSuccess = ApplyPassiveEffectToOwner(PassiveData, TargetSlot, NewSlotInfo.ItemSourceTag); // Передаем тег для применения
            // Для пассивок нам не нужен хэндл эффекта, т.к. удаляем по тегу
        }
        else if (UWeaponSystemItemData* WeaponData = Cast<UWeaponSystemItemData>(ItemToEquip))
        {
            bool bFireGranted = GrantAbilityToOwner(WeaponData, TargetSlot, NewSlotInfo.GrantedAbilityHandle);
            // TODO: Выдать доп. способности (Reload, AltFire) аналогично, если они есть
            // ... GrantAbilityToOwner(WeaponData->ReloadAbilityClass, ...); ...
            // Применяем тег блокировки базовой атаки
            if (bFireGranted && WeaponData->WeaponEquippedTag.IsValid())
            {
                ASC->AddLooseGameplayTag(WeaponData->WeaponEquippedTag);
                WeaponEquippedTags.AddTag(WeaponData->WeaponEquippedTag); // Сохраняем локально
                UE_LOG(LogTemp, Log, TEXT("EquipItem: Added WeaponEquippedTag %s"), *WeaponData->WeaponEquippedTag.ToString());
            }
             bSuccess = bFireGranted; // Успех, если хотя бы основная атака выдана
        }
        // TODO: Добавить обработку других типов предметов (Movement, Support, WeaponMod)

        // --- Обновляем карту слотов и репликацию ---
        if (bSuccess)
        {
            EquippedSlots.Add(TargetSlot, NewSlotInfo);

            return true;
        }
        else

             UE_LOG(LogTemp, Error, TEXT("ChassisSlotManagerComponent::EquipItem: Failed to grant ability/effect for item %s"), *ItemToEquip->GetName());

    UFrameForgeItemData* UChassisSlotManagerComponent::UnequipItem(ESlotType TargetSlot)
    {
        // Выполняем ТОЛЬКО НА СЕРВЕРЕ
        AActor* Owner = GetOwner();
        if (!Owner || Owner->GetLocalRole() != ROLE_Authority)
        {
             UE_LOG(LogTemp, Warning, TEXT("ChassisSlotManagerComponent::UnequipItem called on client for %s. Ignoring."), *GetNameSafe(Owner));
            return nullptr;
        }

        // Находим информацию о слоте
        if (FEquippedSlotInfo* SlotInfoPtr = EquippedSlots.Find(TargetSlot))
        {
            UFrameForgeItemData* UnequippedItemData = SlotInfoPtr->ItemData; // Сохраняем для возврата
            if (!UnequippedItemData) return nullptr; // Если данных нет, выходим

            UAbilitySystemComponent* ASC = GetOwnerASC();
            if (!ASC) return nullptr;

            // --- Удаляем Способность или Эффект ---
            if (SlotInfoPtr->GrantedAbilityHandle.IsValid())
            {
                ClearAbilityFromOwner(TargetSlot); // Использует хэндл из SlotInfoPtr
            }
            else if (SlotInfoPtr->ItemSourceTag.IsValid()) // Если был пассивный эффект (удаляем по тегу)
            {
                RemovePassiveEffectFromOwner(TargetSlot); // Использует тег из SlotInfoPtr
            }

             // --- Снимаем тег оружия, если это было оружие ---
             if (UWeaponSystemItemData* WeaponData = Cast<UWeaponSystemItemData>(UnequippedItemData))
             {
                 if (WeaponData->WeaponEquippedTag.IsValid())
                 {
                     ASC->RemoveLooseGameplayTag(WeaponData->WeaponEquippedTag);
                     WeaponEquippedTags.RemoveTag(WeaponData->WeaponEquippedTag);
                     UE_LOG(LogTemp, Log, TEXT("UnequipItem: Removed WeaponEquippedTag %s"), *WeaponData->WeaponEquippedTag.ToString());
                 }
             }

            // --- Очищаем слот и реплицируем ---
            EquippedSlots.Remove(TargetSlot);
            // Вызываем OnRep вручную на сервере
            // OnRep_EquippedSlots();
             UE_LOG(LogTemp, Log, TEXT("ChassisSlotManagerComponent::UnequipItem: Successfully unequipped item from slot %d"), TargetSlot);

            // TODO: Вызвать делегат или событие о снятии предмета (для визуала, UI)
            // OnItemUnequippedDelegate.Broadcast(TargetSlot, UnequippedItemData);

            return UnequippedItemData; // Возвращаем данные снятого предмета
        }

        // Слот был пуст
        return nullptr;
    }

    bool UChassisSlotManagerComponent::GrantAbilityToOwner(UFrameForgeItemData* ItemData, ESlotType TargetSlot, FGameplayAbilitySpecHandle& OutHandle)
    {
        UAbilitySystemComponent* ASC = GetOwnerASC();
        if (!ASC || !ItemData) return false;

        TSubclassOf<UGameplayAbility> AbilityClass = nullptr;
        EAbilityInputID InputID = EAbilityInputID::None;

        // Определяем класс способности и InputID по типу данных
        if (UModuleAbilityItemData* AbilityData = Cast<UModuleAbilityItemData>(ItemData))
        {
            AbilityClass = AbilityData->GrantedAbilityClass;
            InputID = AbilityData->AbilityInputID;
        }
        else if (UWeaponSystemItemData* WeaponData = Cast<UWeaponSystemItemData>(ItemData))
        {
            AbilityClass = WeaponData->FireAbilityClass; // Выдаем основную атаку
            InputID = EAbilityInputID::PrimaryAttack; // Оружие всегда на PrimaryAttack
            // TODO: Обработать Reload/AltFire Abilities
        }
        // TODO: Добавить другие типы (Movement, Support)

        if (!IsValid(AbilityClass))
        {
             UE_LOG(LogTemp, Warning, TEXT("GrantAbilityToOwner: Invalid AbilityClass for item %s"), *ItemData->GetName());
             return false;
        }

        // Создаем Spec и выдаем способность
        FGameplayAbilitySpec AbilitySpec(AbilityClass, 1, static_cast<int32>(InputID), GetOwner()); // Уровень 1, передаем InputID
        OutHandle = ASC->GiveAbility(AbilitySpec);

        if(OutHandle.IsValid())
        {
            UE_LOG(LogTemp, Log, TEXT("GrantAbilityToOwner: Granted ability %s with InputID %d for item %s"), *AbilityClass->GetName(), static_cast<int32>(InputID), *ItemData->GetName());
            return true;
        }
        else
        {
            UE_LOG(LogTemp, Error, TEXT("GrantAbilityToOwner: Failed to grant ability %s for item %s"), *AbilityClass->GetName(), *ItemData->GetName());
            return false;
        }
    }

    bool UChassisSlotManagerComponent::ApplyPassiveEffectToOwner(UFrameForgeItemData* ItemData, ESlotType TargetSlot, FGameplayTag& OutSourceTag)
    {
        UAbilitySystemComponent* ASC = GetOwnerASC();
        if (!ASC || !ItemData) return false;

        TSubclassOf<UGameplayEffect> EffectClass = nullptr;

        if (UComponentPassiveItemData* PassiveData = Cast<UComponentPassiveItemData>(ItemData))
        {
            EffectClass = PassiveData->GrantedEffectClass;
            OutSourceTag = PassiveData->ItemSourceTag; // Получаем тег из Data Asset
        }
        // TODO: Добавить другие типы (WeaponMod?)

        if (!IsValid(EffectClass))
        {
             UE_LOG(LogTemp, Warning, TEXT("ApplyPassiveEffectToOwner: Invalid EffectClass for item %s"), *ItemData->GetName());
             return false;
        }
        if (!OutSourceTag.IsValid())
        {
             UE_LOG(LogTemp, Error, TEXT("ApplyPassiveEffectToOwner: Invalid ItemSourceTag for item %s! Cannot apply effect reliably."), *ItemData->GetName());
             return false;
        }

        // Создаем контекст и спецификацию
        FGameplayEffectContextHandle EffectContext = ASC->MakeEffectContext();
        EffectContext.AddSourceObject(GetOwner()); // Источник - владелец компонента
        FGameplayEffectSpecHandle SpecHandle = ASC->MakeOutgoingSpec(EffectClass, 1, EffectContext); // Уровень 1

        if (SpecHandle.IsValid())
        {
            // --- Ключевой момент: Добавляем тег источника к спецификации ---
            // Это позволит нам потом удалить эффект именно от этого предмета.
            SpecHandle.Data->DynamicGrantedTags.AddTag(OutSourceTag);

            // Применяем эффект
            FActiveGameplayEffectHandle AppliedHandle = ASC->ApplyGameplayEffectSpecToSelf(*SpecHandle.Data.Get());

            if (AppliedHandle.WasSuccessfullyApplied())
            {
                UE_LOG(LogTemp, Log, TEXT("ApplyPassiveEffectToOwner: Applied effect %s with SourceTag %s for item %s"), *EffectClass->GetName(), *OutSourceTag.ToString(), *ItemData->GetName());
                // Нам не нужно сохранять AppliedHandle, т.к. удаляем по тегу
                return true;
            }
            else
            {
                 UE_LOG(LogTemp, Error, TEXT("ApplyPassiveEffectToOwner: Failed to apply effect %s for item %s"), *EffectClass->GetName(), *ItemData->GetName());
                 return false;
            }
        }
        else
        {
             UE_LOG(LogTemp, Error, TEXT("ApplyPassiveEffectToOwner: Failed to make spec for effect %s for item %s"), *EffectClass->GetName(), *ItemData->GetName());
             return false;
        }
    }

    void UChassisSlotManagerComponent::ClearAbilityFromOwner(ESlotType TargetSlot)
    {
        UAbilitySystemComponent* ASC = GetOwnerASC();
        if (!ASC) return;

        if (FEquippedSlotInfo* SlotInfoPtr = EquippedSlots.Find(TargetSlot))
        {
            if (SlotInfoPtr->GrantedAbilityHandle.IsValid())
            {
                UE_LOG(LogTemp, Log, TEXT("ClearAbilityFromOwner: Clearing ability with handle %s from slot %d"), *SlotInfoPtr->GrantedAbilityHandle.ToString(), TargetSlot);
                ASC->ClearAbility(SlotInfoPtr->GrantedAbilityHandle); // Удаляем по хэндлу
                SlotInfoPtr->GrantedAbilityHandle.Invalidate(); // Сбрасываем хэндл в структуре
            }
        }
    }

    void UChassisSlotManagerComponent::RemovePassiveEffectFromOwner(ESlotType TargetSlot)
    {
        UAbilitySystemComponent* ASC = GetOwnerASC();
        if (!ASC) return;

        if (FEquippedSlotInfo* SlotInfoPtr = EquippedSlots.Find(TargetSlot))
        {
            if (SlotInfoPtr->ItemSourceTag.IsValid())
            {
                FGameplayTagContainer TagToRemoveContainer;
                TagToRemoveContainer.AddTag(SlotInfoPtr->ItemSourceTag);

                int32 NumRemoved = ASC->RemoveActiveGameplayEffectsWithGrantedTags(TagToRemoveContainer); // Удаляем по тегу
                UE_LOG(LogTemp, Log, TEXT("RemovePassiveEffectFromOwner: Removed %d effects with SourceTag %s from slot %d"), NumRemoved, *SlotInfoPtr->ItemSourceTag.ToString(), TargetSlot);

                if(NumRemoved == 0) {
                     UE_LOG(LogTemp, Warning, TEXT("RemovePassiveEffectFromOwner: No effects found with tag %s to remove."), *SlotInfoPtr->ItemSourceTag.ToString());
                }
                SlotInfoPtr->ItemSourceTag = FGameplayTag::EmptyTag; // Сбрасываем тег в структуре
            }
             else
             {
                  UE_LOG(LogTemp, Warning, TEXT("RemovePassiveEffectFromOwner: Item in slot %d had no valid ItemSourceTag."), TargetSlot);
             }
        }
    }
