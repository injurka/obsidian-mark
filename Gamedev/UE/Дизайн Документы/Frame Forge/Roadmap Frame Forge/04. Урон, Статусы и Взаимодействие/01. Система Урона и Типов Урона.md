**Система Урона и Типов Урона**. Мы будем использовать `GameplayEffectExecutionCalculation` (GEEC) для более гибкой и инкапсулированной логики расчета урона, учитывающей типы урона и сопротивления/броню.

**Задача:** Определить GE для различных типов урона, использовать `GameplayTag` для их представления, добавить атрибуты сопротивлений, создать GEEC для расчета финального урона и обновить существующие GE урона для использования этой системы.

**Зачем:** Создать систему, где разные атаки могут наносить урон разного типа (огонь, холод, энергия и т.д.), а цели могут по-разному на него реагировать (сопротивления, броня), делая боевую систему глубже.

**GAS Компоненты и Механизмы UE:** `GameplayTag`, `UAttributeSet`, `UGameplayEffectExecutionCalculation`, `GameplayEffect`, `FGameplayModifierEvaluatedData`, `FAggregatorEvaluateParameters`.

    void UFrameForgeDamageExecCalculation::Execute_Implementation(const FGameplayEffectCustomExecutionParameters& ExecutionParams, FGameplayEffectCustomExecutionOutput& OutExecutionOutput) const
    {
        // --- Получаем необходимые компоненты и данные ---
        UAbilitySystemComponent* TargetASC = ExecutionParams.GetTargetAbilitySystemComponent();
        UAbilitySystemComponent* SourceASC = ExecutionParams.GetSourceAbilitySystemComponent();
        AActor* TargetActor = TargetASC ? TargetASC->GetAvatarActor() : nullptr;
        AActor* SourceActor = SourceASC ? SourceASC->GetAvatarActor() : nullptr;
        const FGameplayEffectSpec& Spec = ExecutionParams.GetOwningSpec();
        const FGameplayTagContainer* SourceTags = Spec.CapturedSourceTags.GetAggregatedTags(); // Теги ИСТОЧНИКА эффекта (важно для крита, типа урона)
        const FGameplayTagContainer* TargetTags = Spec.CapturedTargetTags.GetAggregatedTags(); // Теги ЦЕЛИ (для уязвимостей и т.п.)

        // Параметры для безопасного получения значений атрибутов
        FAggregatorEvaluateParameters EvaluateParameters;
        EvaluateParameters.SourceTags = SourceTags;
        EvaluateParameters.TargetTags = TargetTags;

        // --- Получаем Базовый Урон (из мета-атрибута Damage) ---
        float BaseDamage = 0.0f;
        // Пытаемся получить магнитуду мета-атрибута Damage. SetByCaller используется, если нет модификатора.
        ExecutionParams.AttemptCalculateCapturedAttributeMagnitude(DamageStatics().DamageDef, EvaluateParameters, BaseDamage);
        // Добавляем любые другие базовые модификаторы урона, если они есть в Spec
        BaseDamage += FMath::Max<float>(0.0f, Spec.GetSetByCallerMagnitude(FGameplayTag::RequestGameplayTag(FName("Data.Damage")), false, -1.0f));

        if (BaseDamage <= 0.0f)
        {
            // Нет урона, выходим
            return;
        }

        // --- Определяем Тип Урона ---
        FGameplayTag DamageTypeTag;
        // Ищем первый тег, соответствующий Damage.Type.* в тегах источника
        for (const FGameplayTag& Tag : *SourceTags)
        {
            if (Tag.MatchesTag(FGameplayTag::RequestGameplayTag(FName("Damage.Type")))) // Ищем родительский тег
            {
                 if (Tag != FGameplayTag::RequestGameplayTag(FName("Damage.Type"))) // Убеждаемся, что это не сам родительский тег
                 {
                      DamageTypeTag = Tag;
                      break; // Нашли первый тип, выходим
                 }
            }
        }
        if (!DamageTypeTag.IsValid())
        {
            // Тип урона не указан в тегах источника! Используем Физический по умолчанию или выводим ошибку.
            DamageTypeTag = FGameplayTag::RequestGameplayTag(FName("Damage.Type.Physical"));
             UE_LOG(LogTemp, Warning, TEXT("UFrameForgeDamageExecCalculation: DamageTypeTag not found in SourceTags for spec %s. Defaulting to Physical."), *Spec.Def->GetName());
        }

        // --- Получаем Сопротивление Цели ---
        float Resistance = GetResistanceByDamageTypeTag(*TargetTags, DamageTypeTag, EvaluateParameters, TargetASC);
        Resistance = FMath::Clamp(Resistance, 0.0f, 1.0f); // Зажимаем сопротивление (0-1 для 0-100% снижения)

        // --- Получаем Броню Цели ---
        float TargetArmor = 0.0f;
        ExecutionParams.AttemptCalculateCapturedAttributeMagnitude(DamageStatics().ArmorDef, EvaluateParameters, TargetArmor);
        TargetArmor = FMath::Max(0.0f, TargetArmor); // Броня не может быть отрицательной

        // --- Применяем Сопротивление ---
        // Урон уменьшается на процент сопротивления
        float MitigatedDamage = BaseDamage * (1.0f - Resistance);

        // --- Применяем Броню ---
        // *Решение Дизайна:* Как броня взаимодействует с типами урона?
        // Вариант 1: Броня снижает ТОЛЬКО Физический урон.
        // Вариант 2: Броня снижает ЛЮБОЙ урон ПОСЛЕ сопротивлений (как здесь).
        // Вариант 3: Броня имеет эффективность против разных типов урона. (Сложно)
        // Выбираем Вариант 2 для примера:
        if (DamageTypeTag == FGameplayTag::RequestGameplayTag(FName("Damage.Type.Physical"))) // Пример: Броня работает только против физики
        {
             MitigatedDamage -= TargetArmor;
        }
        // Если броня работает против всего:
        // MitigatedDamage -= TargetArmor;

        // Урон не может быть меньше нуля
        float FinalDamage = FMath::Max(0.0f, MitigatedDamage);

        // --- Обработка Критического Удара ---
        bool bIsCritical = SourceTags->HasTagExact(FGameplayTag::RequestGameplayTag(FName("Damage.CriticalHit")));
        if (bIsCritical)
        {
            // Получаем множитель крита источника (если есть, иначе используем дефолт)
            // float SourceCritDamageMultiplier = 2.0f; // Дефолт
            // ExecutionParams.AttemptCalculateCapturedAttributeMagnitude(DamageStatics().CritDamageDef, EvaluateParameters, SourceCritDamageMultiplier); // Если бы захватывали CritDamage источника

            // Получаем сопротивление криту цели
            float TargetCritResistance = 0.0f;
            ExecutionParams.AttemptCalculateCapturedAttributeMagnitude(DamageStatics().CritDamageResistanceDef, EvaluateParameters, TargetCritResistance);
            TargetCritResistance = FMath::Clamp(TargetCritResistance, 0.0f, 1.0f);

            // Рассчитываем бонусный урон от крита
            // *Решение Дизайна:* Крит умножает урон до или после брони/сопротивлений?
            // Пример: Умножаем ПОСЛЕ. Бонусный урон = FinalDamage * (МножительКрита - 1.0)
            float CritDamageMultiplier = 2.0f; // Базовый множитель +100%
            // TODO: Получить реальный множитель крита из атрибута источника, если нужно
            float BonusCritDamage = FinalDamage * (CritDamageMultiplier - 1.0f);

            // Уменьшаем бонусный урон на сопротивление криту
            BonusCritDamage *= (1.0f - TargetCritResistance);

            // Добавляем бонусный урон
            FinalDamage += FMath::Max(0.0f, BonusCritDamage);

             UE_LOG(LogTemp, Log, TEXT("CRITICAL HIT! Base: %.2f, Mitigated: %.2f, BonusCrit: %.2f, Final: %.2f"), BaseDamage, MitigatedDamage, BonusCritDamage, FinalDamage);
        }

        // --- Вывод Результата ---
        if (FinalDamage > 0.0f)
        {
            // Добавляем модификатор, который будет применен к атрибуту Health ЦЕЛИ.
            // Используем Additive, так как мы вычитаем урон (добавляем отрицательное значение).
            OutExecutionOutput.AddOutputModifier(FGameplayModifierEvaluatedData(DamageStatics().HealthProperty, EGameplayModOp::Additive, -FinalDamage));
             UE_LOG(LogTemp, Log, TEXT("Damage Exec: Type=%s, Base=%.2f, Resist=%.2f, Armor=%.2f, Final=%.2f"), *DamageTypeTag.ToString(), BaseDamage, Resistance, TargetArmor, FinalDamage);
        }
         else
         {
              UE_LOG(LogTemp, Log, TEXT("Damage Exec: Type=%s, Base=%.2f resulted in 0 final damage."), *DamageTypeTag.ToString(), BaseDamage);
         }
    }

    // Вспомогательная функция для получения нужного сопротивления

**Шаг 14.5: Обновление `BaseAttributeSet::PostGameplayEffectExecute`**

Логика обработки урона теперь находится в GEEC. `PostGameplayEffectExecute` в `BaseAttributeSet` должен только реагировать на *результат* изменения `Health`.

1.  **Откройте `BaseAttributeSet.cpp`**.
2.  **Найдите `PostGameplayEffectExecute`**.
3.  **Удалите** или закомментируйте блок, который проверял `if (TargetAttribute == GetDamageAttribute())` и вызывал `HandleDamage`.
4.  **Оставьте** или доработайте блок `else if (TargetAttribute == GetHealthAttribute())`:
    ```cpp
    void UBaseAttributeSet::PostGameplayEffectExecute(const FGameplayEffectModCallbackData& Data)
    {
        Super::PostGameplayEffectExecute(Data);

        FGameplayEffectContextHandle Context = Data.EffectSpec.GetContext();
        UAbilitySystemComponent* SourceASC = Context.GetOriginalInstigatorAbilitySystemComponent();
        const FGameplayTagContainer* SourceTags = Data.EffectSpec.CapturedSourceTags.GetAggregatedTags();
        FGameplayAttribute TargetAttribute = Data.EvaluatedData.Attribute;
        AActor* TargetActor = Data.Target.GetAvatarActor(); // Получаем цель

        // --- Обработка Лечения (если используете мета-атрибут Healing) ---
        if (TargetAttribute == GetHealingAttribute())
        {
           HandleHealing(Data); // Вызываем старую функцию лечения
        }
        // --- Обработка Изменения Здоровья (после расчета урона/лечения в GEEC или HandleHealing) ---
        else if (TargetAttribute == GetHealthAttribute())
        {
             // Получаем старое значение перед изменением (может быть неточным из-за порядка вызовов, лучше передавать)
             // float OldHealth = ???; // Сложно получить надежно здесь
             // Проще получить текущее значение ПОСЛЕ изменения и сравнить с 0

             const float CurrentHealth = GetHealth(); // Получаем УЖЕ измененное значение

             // Зажимаем здоровье еще раз (на всякий случай, хотя GEEC и PreAttributeChange должны это делать)
             SetHealth(FMath::Clamp(CurrentHealth, 0.0f, GetMaxHealth()));

             // --- Проверка Смерти ---
             // Проверяем, если здоровье <= 0 И цель еще не помечена как мертвая
             if (CurrentHealth <= 0.0f && TargetActor && !Data.Target.HasMatchingGameplayTag(FGameplayTag::RequestGameplayTag(FName("State.Dead"))))
             {
                 // Отправляем событие смерти
                 FGameplayEventData Payload;
                 Payload.EventTag = FGameplayTag::RequestGameplayTag(FName("Event.State.Dead"));
                 Payload.Instigator = SourceASC ? SourceASC->GetAvatarActor() : nullptr; // Кто нанес урон
                 Payload.Target = TargetActor; // Кто умер
                 Payload.ContextHandle = Context; // Передаем контекст
                 Payload.TargetTags = *TargetTags;  // Теги цели
                 Payload.EventMagnitude = Data.EvaluatedData.Magnitude; // Величина изменения здоровья

                 // Отправляем событие в ASC цели
                 UAbilitySystemComponent* TargetASC = UAbilitySystemBlueprintLibrary::GetAbilitySystemComponent(TargetActor);
                 if(TargetASC)
                 {
                     TargetASC->HandleGameplayEvent(Payload.EventTag, &Payload);
                      UE_LOG(LogTemp, Log, TEXT("PostGameplayEffectExecute: Sent Death event for %s"), *TargetActor->GetName());
                 }

                 // Добавляем тег State.Dead к цели (важно, чтобы предотвратить повторную обработку смерти)
                 // Лучше делать это в способности, которая реагирует на Event.State.Dead
                 // TargetASC->AddLooseGameplayTag(FGameplayTag::RequestGameplayTag(FName("State.Dead")));
             }
        }
        // ... (обработка Energy, Stamina, если нужно) ...
    }
    ```
    *Ключевое изменение:* Мы больше не обрабатываем мета-атрибут `Damage` здесь. Мы реагируем на фактическое изменение `Health` (которое уже было рассчитано GEEC) и проверяем условие смерти.
