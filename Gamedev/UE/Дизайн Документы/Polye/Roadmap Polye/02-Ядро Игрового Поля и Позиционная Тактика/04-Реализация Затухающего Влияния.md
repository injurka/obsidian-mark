        ```cpp
        // SERVER ONLY - Рассчитывает влияние и применяет эффекты
        void APolyeGameStateBase::CalculateAndApplyInfluence()
        {
             if (!HasAuthority()) return; // Только сервер рассчитывает

             UE_LOG(LogTemp, Log, TEXT("Расчет и Применение Влияния..."));

             // Временная мапа для построения *нового* состояния влияния
             TMap<int32, FActiveInfluenceOnField> NewInfluenceState;

             // 1. Определить все поля-источники (Активные поля с Фракциями)
             TArray<int32> SourceFieldIndices;
             for (const FPolyeFieldState& State : BoardState)
             {
                  if (State.bIsActive && State.CurrentFaction != EPolyeFaction::EF_Neutral)
                  {
                       SourceFieldIndices.Add(State.FieldIndex);
                  }
             }

             // Получаем доступ к Менеджеру Фракций (предполагаем, он доступен через GameInstance)
             UPolyeFactionManager* FactionManager = GetGameInstance() ? Cast<UPolyeFactionManager>(GetGameInstance()->GetFactionManager()) : nullptr; // Пример геттера в UPolyeGameInstance
             if (!FactionManager)
             {
                  UE_LOG(LogTemp, Error, TEXT("Менеджер Фракций не найден!"));
                  return;
             }

             // 2. Для каждого поля-источника определить затронутые поля и потенциальные ауры
             // Map: FieldIndex -> SourceFaction -> AuraType -> Strength
             TMap<int32, TMap<EPolyeFaction, TMap<EPolyeEffectType, float>>> PotentialInfluencePerField;

             for (int32 SourceIndex : SourceFieldIndices)
             {
                  const FPolyeFieldState& SourceState = BoardState[SourceIndex];
                  EPolyeFaction SourceFaction = SourceState.CurrentFaction;

                  // Получаем данные Фракции, чтобы знать, какие Ауры эта фракция испускает
                  const UPolyeFactionData* FactionData = FactionManager->GetFactionData(SourceFaction); // Реализовать этот геттер в FactionManager
                  if (!FactionData) continue;

                  // Определяем затронутые поля на расстоянии 50% и 25%
                  TArray<int32> Fields50Percent = UPolyeBoardStatics::GetAdjacentFields(SourceIndex, false); // false для без диагоналей по умолчанию
                  TArray<int32> Fields25Percent = UPolyeBoardStatics::GetFieldsTwoStepsAway(SourceIndex, false); // false для без диагоналей по умолчанию

                  // Применяем Ауры к полю-источнику (100%) и затронутым полям (50%, 25%)
                  TArray<int32> AllAffectedIndices;
                  AllAffectedIndices.Add(SourceIndex); // 100%
                  AllAffectedIndices.Append(Fields50Percent); // 50%
                  AllAffectedIndices.Append(Fields25Percent); // 25%

                  for (int32 AffectedIndex : AllAffectedIndices)
                  {
                       // Только применять влияние к *активным* полям
                       if (!BoardState.IsValidIndex(AffectedIndex) || !BoardState[AffectedIndex].bIsActive)
                       {
                            continue;
                       }

                       float Strength = 0.0f;
                       if (AffectedIndex == SourceIndex) Strength = 1.0f;
                       else if (Fields50Percent.Contains(AffectedIndex)) Strength = 0.5f;
                       else if (Fields25Percent.Contains(AffectedIndex)) Strength = 0.25f;
                       else continue; // Не должно случиться при корректной логике соседей

                       // Записываем потенциальное влияние от этой фракции-источника
                       for (const FDecayingInfluenceAuraConfig& AuraConfig : FactionData->DecayingInfluenceAuras)
                       {
                             // TODO: Проверить, иммунна ли Аура к себе на поле источника (по дизайну Братства Топей?)
                             // If (AffectedIndex == SourceIndex && AuraConfig.AuraType == EPolyeEffectType::PET_DecayingInfluence_HealReduction && SourceFaction == EPolyeFaction::EF_SwampBrotherhood) continue; // Пример иммунитета

                            PotentialInfluencePerField.FindOrAdd(AffectedIndex).FindOrAdd(SourceFaction).FindOrAdd(AuraConfig.AuraType) = Strength;
                       }
                  }
             }

             // 3. Для каждого активного поля определить *эффективные* Ауры (применить правила сложения)
             for (const FPolyeFieldState& State : BoardState)
             {
                  if (!State.bIsActive) continue;

                  TArray<FEffectiveAuraState> EffectiveAurasForField;
                  if (PotentialInfluencePerField.Contains(State.FieldIndex))
                  {
                       const TMap<EPolyeFaction, TMap<EPolyeEffectType, float>>& InfluenceFromAllSources = PotentialInfluencePerField[State.FieldIndex];

                       // Map для отслеживания максимальной силы для каждого Типа Ауры
                       TMap<EPolyeEffectType, float> MaxStrengthPerAuraType;

                       // Правило: "От одной фракции: ... применяется только самый сильный эффект от этой конкретной Ауры."
                       // Это подразумевает, что для данного Типа Ауры (напр., HealReduction), от данной Фракции (напр., SwampBrotherhood),
                       // важна только максимальная сила ОТ ЭТОЙ фракции. Но Ауры разных фракций складываются.
                       // Пример: Поле рядом с БТ (50% СнижениеЛечения), и через 2 поля от КС (25% ПериодическийУрон). Результат: 50% СнижениеЛечения + 25% ПериодическийУрон.
                       // Пример: Поле рядом с БТ1 (50% СнижениеЛечения), и через 2 поля от БТ2 (25% СнижениеЛечения). Результат: Макс(50%, 25%) = 50% СнижениеЛечения.

                       // Реализуем правило: для каждого УНИКАЛЬНОГО Типа Ауры, найти МАКСИМАЛЬНУЮ силу от *любой* отдельной фракции, которая ее проецирует.
                       TMap<EPolyeEffectType, float> MaxStrengthOverallPerAuraType;

                       for (const auto& PairFactionInfluence : InfluenceFromAllSources) // Итерация по фракциям-источникам
                       {
                            const TMap<EPolyeEffectType, float>& InfluenceByAuraType = PairFactionInfluence.Value; // Получаем Влияние от одной фракции-источника

                            for (const auto& PairAuraStrength : InfluenceByAuraType) // Итерация по Аурам от этой фракции
                            {
                                 EPolyeEffectType AuraType = PairAuraStrength.Key;
                                 float Strength = PairAuraStrength.Value;

                                 // Обновляем максимальную силу, найденную для этого Типа Ауры
                                 if (!MaxStrengthOverallPerAuraType.Contains(AuraType) || Strength > MaxStrengthOverallPerAuraType[AuraType])
                                 {
                                      MaxStrengthOverallPerAuraType.Add(AuraType, Strength);
                                 }
                             }
                       }

                       // Компилируем финальный список эффективных аур для этого поля
                       for (const auto& PairAuraStrength : MaxStrengthOverallPerAuraType)
                       {
                            FEffectiveAuraState EffectiveAura;
                            EffectiveAura.AuraType = PairAuraStrength.Key;
                            EffectiveAura.Strength = PairAuraStrength.Value;

                            // TODO: Получить FPolyeEffectConfig с *параметрами*, соответствующими этой силе влияния
                            // Например, для СниженияЛечения: 50% = -2, 25% = -1.
                            // Эта логика должна быть в FactionManager или отдельной фабрике эффектов.
                            // UPolyeFactionManager::GetEffectConfigForAura(NewAura.AuraType, NewAura.Strength);
                            // Для MVP можно временно использовать хардкод или простую мапу:
                            if (const FDecayingInfluenceAuraConfig* AuraConfigData = FactionManager->GetAuraConfigData(EffectiveAura.AuraType)) // Нужен геттер в FactionManager
                            {
                                 // Создать FPolyeEffectConfig с нужными параметрами
                                 EffectiveAura.EffectConfig = *AuraConfigData->GetConfigForStrength(EffectiveAura.Strength); // Нужна функция в FDecayingInfluenceAuraConfig для получения конфига по силе
                            }
                            else
                            {
                                 UE_LOG(LogTemp, Error, TEXT("Configuration for Aura Type %s not found!"), *UEnum::GetValueAsString(EffectiveAura.AuraType));
                                 continue; // Пропускаем, если нет конфига
                            }

                            EffectiveAurasForField.Add(EffectiveAura);
                       }
                  }

                  // Сохраняем рассчитанное состояние
                  NewInfluenceState.Add(State.FieldIndex, FActiveInfluenceOnField()); // Добавляем обертку структуры
                  NewInfluenceState[State.FieldIndex].EffectiveAuras = EffectiveAurasForField;
             }

             // 4. Сравниваем старое состояние с новым и применяем/удаляем эффекты на сущностях
             // Эта логика сравнения критически важна для производительности и корректности.
             // Итерируем по всем полям.

             TMap<int32, FActiveInfluenceOnField> OldInfluenceState = ActiveInfluenceState; // Копируем старое состояние

             // Обновляем реплицируемое состояние *до* применения эффектов - визуал клиентов реагирует на это
             ActiveInfluenceState = NewInfluenceState;
             MarkNetDirty(); // Убеждаемся, что обновленная мапа реплицируется

             // Теперь применяем/удаляем эффекты на основе разницы
             for (int32 i = 0; i < BoardState.Num(); ++i) // Итерируем по всем полям
             {
                 const FPolyeFieldState& FieldState = BoardState[i];
                 // Получаем новое и старое эффективное влияние для этого поля
                 const FActiveInfluenceOnField NewFieldInfluence = NewInfluenceState.Contains(i) ? NewInfluenceState[i] : FActiveInfluenceOnField();
                 const FActiveInfluenceOnField OldFieldInfluence = OldInfluenceState.Contains(i) ? OldInfluenceState[i] : FActiveInfluenceOnField();

                 // Если поле было активно и стало неактивным, удаляем все эффекты влияния
                 if (!FieldState.bIsActive && OldInfluenceState.Contains(i)) // OldInfluenceState содержит только активные поля
                 {
                      for (const FEffectiveAuraState& OldAura : OldFieldInfluence.EffectiveAuras)
                      {
                          // Удаляем эффекты, соответствующие OldAura, с сущностей на этом поле (на сервере!)
                          for (ABoardEntityBase* Entity : OldInfluenceState[i].EntitiesOnField) // Итерируем по сущностям, которые были на поле в старом состоянии
                          {
                              if (Entity && !Entity->IsPendingKill() && Entity->GetEffectManagerComponent())
                              {
                                   // TODO: Нужен способ найти и удалить именно тот экземпляр эффекта, который был применен этой аурой
                                   // Сейчас удаляем все эффекты с таким типом AuraType. Это может быть неправильно, если эффект
                                   // с таким же типом применен от другого источника (напр., карты).
                                   FPolyeEffectConfig DummyConfig; // Создаем фиктивный конфиг только для типа
                                   DummyConfig.EffectType = OldAura.AuraType;
                                  Entity->GetEffectManagerComponent()->RemoveEffectsByConfig(DummyConfig); // Нуждается в доработке
                              }
                          }
                      }
                     continue; // На неактивных полях эффекты влияния не действуют
                 }

                 if (!FieldState.bIsActive) continue; // Игнорируем поля, которые не активны в новом состоянии

                  // Сравниваем списки EffectiveAuras и применяем/удаляем эффекты
                  // Проходим по старому списку, чтобы найти удаленные Ауры
                  for (const FEffectiveAuraState& OldAura : OldFieldInfluence.EffectiveAuras)
                  {
                       bool bStillActive = false;
                       for (const FEffectiveAuraState& NewAura : NewFieldInfluence.EffectiveAuras)
                       {
                            if (OldAura.AuraType == NewAura.AuraType) // Если тип ауры все еще присутствует
                            {
                                // TODO: Проверить, изменилась ли сила влияния? Если да, возможно, нужно переприменить.
                                bStillActive = true; // Простой вариант: если тип есть в новом списке - не удаляем
                                break;
                            }
                       }
                       if (!bStillActive) // Если аура из старого списка не найдена в новом
                       {
                            // Удалить этот специфический эффект OldAura с сущностей на этом поле
                             for (ABoardEntityBase* Entity : FieldState.EntitiesOnField) // Итерируем по текущим сущностям на поле
                             {
                                  if (Entity && !Entity->IsPendingKill() && Entity->GetEffectManagerComponent())
                                   {
                                       // TODO: Нужно удалить экземпляр эффекта, соответствующий OldAura.EffectConfig
                                       FPolyeEffectConfig DummyConfig;
                                       DummyConfig.EffectType = OldAura.AuraType; // Используем тип как идентификатор
                                       Entity->GetEffectManagerComponent()->RemoveEffectsByConfig(DummyConfig); // Нуждается в доработке для уникальных инстансов
                                   }
                             }
                       }
                  }

                  // Проходим по новому списку, чтобы найти новые или измененные Ауры
                  for (const FEffectiveAuraState& NewAura : NewFieldInfluence.EffectiveAuras)
                  {
                       bool bWasAlreadyActive = false;
                        // float OldStrength = 0.0f; // Если отслеживаем изменение силы
                       for (const FEffectiveAuraState& OldAura : OldFieldInfluence.EffectiveAuras)
                       {
                            if (NewAura.AuraType == OldAura.AuraType)
                            {
                                 bWasAlreadyActive = true;
                                 // OldStrength = OldAura.Strength;
                                break;
                            }
                       }

                       // Если Ауры не было раньше, или если сила изменилась (решите, требует ли изменение силы переприменения)
                       // Простой вариант: если Ауры не было, или тип Ауры есть в новом списке, но не в старом, или конфиг изменился
                       // Более точно: если старый и новый EffectiveAuraState для данного типа не равны
                       bool bStateChanged = !bWasAlreadyActive || (bWasAlreadyActive && OldFieldInfluence.EffectiveAuras[OldFieldInfluence.EffectiveAuras.IndexOfByPredicate([&](const FEffectiveAuraState& Aura){ return Aura.AuraType == NewAura.AuraType; })] != NewAura); // Сложное сравнение

                       if (bStateChanged)
                       {
                            // Удалить старые экземпляры этого типа ауры перед применением новых, если они были
                            if (bWasAlreadyActive)
                            {
                                 for (ABoardEntityBase* Entity : FieldState.EntitiesOnField)
                                  {
                                       if (Entity && !Entity->IsPendingKill() && Entity->GetEffectManagerComponent())
                                        {
                                            FPolyeEffectConfig DummyConfig;
                                            DummyConfig.EffectType = NewAura.AuraType; // Используем тип
                                           Entity->GetEffectManagerComponent()->RemoveEffectsByConfig(DummyConfig); // Нуждается в доработке
                                        }
                                  }
                            }

                            // Применить новый эффект, соответствующий NewAura, к сущностям на этом поле
                             for (ABoardEntityBase* Entity : FieldState.EntitiesOnField)
                             {
                                  if (Entity && !Entity->IsPendingKill() && Entity->GetEffectManagerComponent())
                                  {
                                       // TODO: Нужно получить класс UPolyeGameplayEffectBase для этого типа Ауры
                                       TSubclassOf<UPolyeGameplayEffectBase> EffectClass = FactionManager->GetEffectClassForAuraType(NewAura.AuraType); // Нужен геттер в FactionManager
                                       // FPolyeEffectConfig NewAuraConfig = NewAura.EffectConfig; // Берем уже подготовленный конфиг из FEffectiveAuraState

                                       if (EffectClass)
                                       {
                                            Entity->GetEffectManagerComponent()->ApplyEffect(EffectClass, NewAura.EffectConfig); // Применяем эффект с готовым конфигом
                                       }
                                  }
                             }
                       }
                  }
             } // Конец итерации по полям
        }
        ```

        *   Конкретный класс `UPolyeGameplayEffectBase` (напр., `UPolyeEffect_HealReduction_Class`).
        *   Конкретные *параметры* эффекта (`FPolyeEffectConfig`) для этой силы (напр., для 50% Ауры Снижения Лечения, параметр "ReductionAmount" должен быть 2; для 25% - 1). **Сложность и Альтернативы (без GAS):**

*   **Кастомная Система Эффектов:** Реализация собственной системы (Компонент, владеющий UObject'ами, базовый класс эффекта, ручное применение/удаление, модификация атрибутов через виртуальные функции) сложна в построении, особенно с учетом сетевой синхронизации, управления состоянием (длительность, стеки, источники) и производительности (управление Tick). Это требует значительных усилий для достижения надежности.
*   **Репликация `TArray<TObjectPtr<UObject>>`:** Хотя возможна, требует аккуратной настройки владения и может быть менее производительной, чем репликация более простых типов или структур, или использование GameplayTags для статусов. `OnRep_ActiveEffects` требует ручной логики для определения разницы между старым и новым списком и запуска визуала.
*   **Производительность Расчета Влияния:** Для 15 полей расчет источников, целей и силы влияния, вероятно, будет быстрым. Основное влияние на производительность оказывает итерация по сущностям и применение/удаление эффектов. Это должно быть оптимизировано, если на доске много сущностей.
*   **Параметры Эффектов:** Проектирование `FPolyeEffectConfig` и того, как множители силы применяются к ее параметрам, критически важно и сильно зависит от конкретных эффектов. Мапа параметров `TMap<FName, float>` гибка, но требует аккуратной обработки в каждой реализации эффекта.
*   **Маппинг Ауры на Класс Эффекта/Конфиг:** Логика lookup (напр., в `UPolyeFactionManager`) должна быть надежной и предоставлять правильный класс `UPolyeGameplayEffectBase` и его параметры (`FPolyeEffectConfig`) на основе типа Ауры и силы влияния.
